Automatic Gmail Dispatch Setup

Connect your 1-time Google Script Webhook so clicking Send sends personalized emails automatically directly from your Gmail account.

Quick Setup Guide:

  1. Open script.google.com and click New Project.
  2. Paste the Google Apps Script code below and click Deploy > New Deployment.
  3. Select Web App (Execute as: Me, Access: Anyone), copy the URL, and paste it above!
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var recipients = data.recipients;
  var subject = data.subject;
  var htmlBody = data.htmlBody;
  
  recipients.forEach(function(r) {
    var personalizedBody = htmlBody.replace("Dear Valued Client,", "Dear " + r.name + ",");
    GmailApp.sendEmail(r.email, subject, "", {
      name: "Mark Isaac Ison - Pru Life UK",
      htmlBody: personalizedBody
    });
  });
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
